Audition on your actual job

Each candidate gets a private copy in Postgres. Every tool call is a row, streamed live with Realtime. Checks read what the agent changed, not what it said.

Supabase Postgres + Realtime
lib/runtime/tryout.ts
async function runOne(tryoutId: string, agent: MarketAgent, job: string) {
  const worldId = await createWorld(tryoutId); // a private copy in Postgres

  const onStep = async (step: Step) => {
    // one row per tool call; Realtime streams it to the dashboard
    await db.from("tryout_steps").insert({ tryout_id: tryoutId, ...step });
  };

  await runAgent(agent, job, { worldId, builder: agent.builder }, onStep);

  // Score what the agent changed, not what it said
  const checks = await checksFor(agent.role, worldId, job);
  const verdict = await judge(agent, job, steps, reply);
  const score = 7 * passedShare(checks) + 3 * (verdict.score / 10);
  await db.from("tryouts")
    .update({ status: "scored", score, checks })
    .eq("id", tryoutId);
}
